### Task 2.2 — 通用搜索层 searcher.py（1d）

**目标**：封装 DuckDuckGo 搜索能力，支持通用搜索和 site: 限定搜索。

#### 3.2.1 文件结构

```python
# app/searcher.py
"""
通用搜索层 —— 基于 DuckDuckGo 的网页搜索
支持：
  - 通用搜索（默认）
  - site: 限定搜索（用于中文社媒等特定域名）
  - 结果去重（按 domain 聚类）
  - 相关性排序
"""

from dataclasses import dataclass, field
from enum import Enum
from duckduckgo_search import DDGS
from typing import Optional
import re

class SearchSource(Enum):
    GENERAL = "general_web"       # 通用网页
    SOCIAL = "social"             # 社交媒体
    OFFICIAL = "official"         # 官方网站
    COMMERCIAL = "commercial"     # 商业数据平台

@dataclass
class SearchResult:
    url: str
    title: str
    snippet: str                  # 搜索引擎返回的摘要
    domain: str                   # 提取自 URL 的域名
    source: SearchSource = SearchSource.GENERAL
    relevance_score: float = 0.0  # 0-1 相关性评分

class Searcher:
    """DuckDuckGo 搜索封装"""

    def __init__(self, timeout: int = 10):
        self._timeout = timeout
        self._ddgs = DDGS(timeout=timeout)

    def search(
        self,
        query: str,
        max_results: int = 5,
        region: str = "wt-wt",   # 无区域限制
    ) -> list[SearchResult]:
        """
        通用搜索

        Args:
            query: 搜索关键词
            max_results: 最大结果数
            region: 区域代码（wt-wt=无限制, cn-zh=中国）

        Returns:
            SearchResult 列表（已去重、已排序）
        """
        raw_results = self._ddgs.text(query, max_results=max_results, region=region)
        results = [
            SearchResult(
                url=r["href"],
                title=r["title"],
                snippet=r["body"],
                domain=self._extract_domain(r["href"]),
            )
            for r in raw_results
        ]
        return self._dedup_and_sort(results)

    def site_search(
        self,
        query: str,
        site: str,
        max_results: int = 5,
    ) -> list[SearchResult]:
        """
        Site: 限定搜索

        Args:
            query: 搜索关键词
            site: 域名（如 "zhihu.com"、"xiaohongshu.com"）
            max_results: 最大结果数

        Returns:
            仅来自指定域名的 SearchResult 列表
        """
        # 构造 site: 查询
        site_query = f"{query} site:{site}"
        raw_results = self._ddgs.text(site_query, max_results=max_results * 2)  # 多搜一些，因为 site: 可能不准
        results = []
        seen_domains = set()

        for r in raw_results:
            domain = self._extract_domain(r["href"])
            # 二次过滤：确保真的来自目标域名
            if site in domain and domain not in seen_domains:
                seen_domains.add(domain)
                results.append(SearchResult(
                    url=r["href"],
                    title=r["title"],
                    snippet=r["body"],
                    domain=domain,
                    source=self._classify_source(domain),
                ))
            if len(results) >= max_results:
                break

        return results

    # ===== 内部方法 =====

    def _extract_domain(self, url: str) -> str:
        """从 URL 提取主域名"""
        match = re.match(r"https?://(?:www\.)?([^/]+)", url)
        return match.group(1) if match else url

    def _classify_source(self, domain: str) -> SearchSource:
        """根据域名判断来源类型"""
        social_domains = {
            "xiaohongshu.com", "xhslink.com",           # 小红书
            "zhihu.com", "zhuanlan.zhihu.com",          # 知乎
            "weibo.com",                                # 微博
            "bilibili.com", "b23.tv",                   # B站
            "maimai.cn",                                # 脉脉
            "ts.21cn.pro", "ts.21cn.com",               # 黑猫投诉
            "douyin.com",                               # 抖音
        }
        official_patterns = (                           # 官网特征
            "careers", "jobs", "about", "pricing", "plans"
        )
        commercial_domains = {
            "tianyancha.com", "qcc.com",                # 天眼查 / 企查查
            "qimai.cn",                                 # 七麦
            "crunchbase.com",                           # Crunchbase
        }

        if domain in social_domains or any(d in domain for d in social_domains):
            return SearchSource.SOCIAL
        if domain in commercial_domains:
            return SearchSource.COMMERCIAL
        if any(p in domain for p in official_patterns):
            return SearchSource.OFFICIAL
        return SearchSource.GENERAL

    def _dedup_and_sort(self, results: list[SearchResult]) -> list[SearchResult]:
        """按 domain 去重 + 按相关性排序"""
        # 去重：同一 domain 只保留最好的
        seen = {}
        for r in results:
            if r.domain not in seen or len(r.snippet) > len(seen[r.domain].snippet):
                seen[r.domain] = r
        # 简单排序：标题匹配度（更精细的排序在 scene_search.py 做）
        return list(seen.values())
```

### Task 2.3 — 抓取层 fetcher.py（1d）

**目标**：从 URL 提取正文内容，支持定价页定向提取。

#### 3.3.1 文件结构

```python
# app/fetcher.py
"""
网页内容提取层
支持：
  - 通用正文提取（trafilatura）
  - 定价页定向提取（04 P0 定价监控核心需求）
  - 并发抓取（httpx 异步）
  - Jina Reader fallback
"""

import asyncio
import re
from dataclasses import dataclass
from typing import Optional
import httpx
import trafilatura
from app.config import settings
from app.logger import log

@dataclass
class FetchedPage:
    url: str
    title: str
    text: str                              # 正文（trafilatura 提取）
    publish_date: Optional[str] = None
    word_count: int = 0
    is_pricing_page: bool = False          # 是否为定价页
    pricing_data: Optional[dict] = None    # 定价页结构化数据（如有）

class Fetcher:
    """网页内容抓取器"""

    HEADERS = {
        "User-Agent": (
            "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
            "AppleWebKit/537.36 (KHTML, like Gecko) "
            "Chrome/120.0.0.0 Safari/537.36"
        ),
        "Accept": "text/html,application/xhtml+xml",
        "Accept-Language": "zh-CN,zh;q=0.9,en;q=0.8",
    }

    PRICING_PATTERNS = [
        r"/pricing", r"/price", r"/plans", r"/费用", r"/价格",
        r"/pricing\.html", r"/plans\.html",
    ]

    async def fetch_one(self, url: str) -> Optional[FetchedPage]:
        """抓取单个页面"""
        try:
            async with httpx.AsyncClient(
                headers=self.HEADERS,
                timeout=settings.fetch_timeout,
                follow_redirects=True,
            ) as client:
                resp = await client.get(url)
                resp.raise_for_status()
                html = resp.text

            # 用 trafilatura 提取正文
            extracted = trafilatura.extract(
                html,
                include_comments=False,
                include_tables=True,        # 保留表格（定价页需要）
                favor_precision=True,       # 更精确（宁可少取不多取）
                output_format="text",
            )

            if not extracted or len(extracted.strip()) < settings.fetch_min_word_count:
                log.debug(f"页面内容过短或提取失败: {url}")
                return None

            is_pricing = self._is_pricing_page(url)
            pricing_data = None
            if is_pricing:
                pricing_data = self._extract_pricing_data(html, extracted)

            return FetchedPage(
                url=url,
                title=trafilatura.extract_title(html) or "",
                text=extracted,
                publish_date=trafilatura.extract_date(html),
                word_count=len(extracted),
                is_pricing_page=is_pricing,
                pricing_data=pricing_data,
            )

        except httpx.HTTPStatusError as e:
            log.warning(f"HTTP 错误 {e.response.status_code}: {url}")
            return None
        except Exception as e:
            log.error(f"抓取异常: {url} - {e}")
            return None

    async def fetch(self, urls: list[str], max_pages: int = 3) -> list[FetchedPage]:
        """并发抓取多个页面"""
        tasks = [self.fetch_one(url) for url in urls[:max_pages]]
        results = await asyncio.gather(*tasks, return_exceptions=True)
        pages = [r for r in results if isinstance(r, FetchedPage)]
        log.info(f"抓取完成: {len(pages)}/{len(urls[:max_pages])} 成功")
        return pages

    async def fetch_pricing_page(self, base_url: str) -> Optional[FetchedPage]:
        """
        定价页定向抓取
        尝试常见的定价页 URL 路径
        """
        pricing_paths = ["/pricing", "/price", "/plans", "/pricing.html"]
        for path in pricing_paths:
            url = base_url.rstrip("/") + path
            page = await self.fetch_one(url)
            if page and page.is_pricing_page:
                log.info(f"找到定价页: {url}")
                return page
        log.warning(f"未找到定价页: {base_url}")
        return None

    # ===== 内部方法 =====

    def _is_pricing_page(self, url: str) -> bool:
        """判断是否为定价页"""
        url_lower = url.lower()
        return any(p in url_lower for p in self.PRICING_PATTERNS)

    def _extract_pricing_data(self, html: str, text: str) -> Optional[dict]:
        """
        从定价页提取结构化数据
        MVP 版本：简单启发式提取（后续可升级为 LLM 结构化）
        """
        # TODO: 实现 HTML 表格解析或 LLM 辅助提取
        # 当前返回原始文本，标记为定价相关
        return {"raw_text": text[:2000], "extraction_method": "raw"}
```

```python
"""
中文数据源层 —— 满足 G6（中文互联网优先）和 G1（不编造核实渠道）红线

数据源优先级：

MVP 目标：覆盖 21 项数据源中的 ~10 项（通过直接+间接方式）

from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from enum import Enum
from typing import Optional
import json
import httpx

class SourceType(Enum):
    """数据源类型枚举（全量 21 项的分类）"""
    # 通用

@dataclass
class DataPoint:
    """统一数据点 —— 所有数据源的输出归一化"""
    title: str
    content: str
    url: Optional[str] = None
    source_type: SourceType = SourceType.GENERAL_WEB

# ===== 数据源抽象 =====

class DataSource(ABC):
    """数据源统一抽象接口"""

    @abstractmethod
    def name(self) -> str: ...

    @abstractmethod
    def source_type(self) -> SourceType: ...

    @abstractmethod
    def is_available(self) -> bool: ...

    @abstractmethod
    async def search(self, query: str, **kwargs) -> list[DataPoint]: ...

class TianYanchaSource(DataSource):
    """
    天眼查 —— 融资/工商变更/股东结构/法律风险/知识产权

    接入方式：天眼查开放平台 REST API
    文档：https://open-api.tianyancha.com/
    成本：免费（有调用频率限制，需注册获取 API Key）
    覆盖维度：
      - 融资历史（轮次/金额/日期/投资方）
      - 工商变更（法人/注册资本/经营范围）
      - 司法风险（诉讼/被执行/经营异常）
      - 知识产权（商标/专利/软著）
    对应红线：G1 核实渠道（融资/工商数据必须可核实）
    """

    BASE_URL = "https://open-api.tianyancha.com/services/open/v2"

    def name(self) -> str: return "天眼查"
    def source_type(self) -> SourceType: return SourceType.COMMERCIAL_TIANYANCHA

    def is_available(self) -> bool:
        return bool(settings.tianyancha_api_key)

    async def search(self, query: str, **kwargs) -> list[DataPoint]:
        if not self.is_available():
            log.debug("天眼查未配置 API Key，跳过")
            return []

        results = []
        # 1. 搜索企业基本信息
        company_info = await self._search_company(query)
        if company_info:
            results.append(company_info)

        # 2. 搜索融资历史
        financing = await self._search_financing(query)
        results.extend(financing)

        return results

    async def _search_company(self, company_name: str) -> Optional[DataPoint]:
        """搜索企业基本信息"""
        try:
            async with httpx.AsyncClient(timeout=10) as client:
                resp = await client.get(
                    f"{self.BASE_URL}/companyBaseInfo.json",
                    params={"name": company_name},
                    headers={"Authorization": settings.tianyancha_api_key},
                )
                resp.raise_for_status()
                data = resp.json()

                if data.get("error_code") != 0 or not data.get("data"):
                    return None

                info = data["data"][0]
                content = (
                    f"公司名：{info.get('name', 'N/A')}\n"
                    f"成立时间：{info.get('estiblishTime', 'N/A')}\n"
                    f"法定代表人：{info.get('legalPersonName', 'N/A')}\n"
                    f"注册资本：{info.get('regCapital', 'N/A')}\n"
                    f"经营状态：{info.get('regStatus', 'N/A')}\n"
                    f"经营范围：{info.get('businessScope', 'N/A')[:200]}"
                )
                return DataPoint(
                    title=f"{info.get('name', company_name)} - 企业基本信息",
                    content=content,
                    source_type=SourceType.COMMERCIAL_TIANYANCHA,
                    source_name="天眼查",
                    confidence="high",
                    metadata={"api_endpoint": "companyBaseInfo"},
                )
        except Exception as e:
            log.warning(f"天眼查企业信息查询失败: {company_name} - {e}")
            return None

    async def _search_financing(self, company_name: str) -> list[DataPoint]:
        """搜索融资历史"""
        try:
            async with httpx.AsyncClient(timeout=10) as client:
                resp = await client.get(
                    f"{self.BASE_URL}/financingV2.json",
                    params={"name": company_name},
                    headers={"Authorization": settings.tianyancha_api_key},
                )
                resp.raise_for_status()
                data = resp.json()

                if data.get("error_code") != 0 or not data.get("data"):
                    return []

                results = []
                for item in data["data"][:5]:  # 取最近 5 轮
                    round_name = item.get("round", "N/A")
                    amount = item.get("amount", "N/A")
                    invest_date = item.get("investDate", "N/A")
                    investors = ", ".join(item.get("investorList", []))

                    content = (
                        f"融资轮次：{round_name}\n"
                        f"融资金额：{amount}\n"
                        f"投资时间：{invest_date}\n"
                        f"投资方：{investors or '未披露'}"
                    )
                    results.append(DataPoint(
                        title=f"{company_name} - {round_name}轮融资",
                        content=content,
                        source_type=SourceType.COMMERCIAL_TIANYANCHA,
                        source_name="天眼查",
                        published_at=invest_date,
                        confidence="high",
                        metadata={"api_endpoint": "financingV2"},
                    ))
                return results
        except Exception as e:
            log.warning(f"天眼查融资查询失败: {company_name} - {e}")
            return []


class SiteSearchSource(DataSource):
    """
    Site: 限定搜索 —— 通过 DuckDuckGo site: 操作符搜索特定中文平台

    覆盖平台（MVP）：
      · 小红书 (xiaohongshu.com)
      · 知乎 (zhihu.com)
      · 黑猫投诉 (ts.21cn.pro)
      · 微信公众号（通过搜狗微信搜索 weixin.sogou.com）

    成本：零（复用 DuckDuckGo）
    局限性：只能获取公开可索引的内容
    """

    SITE_CONFIGS = {
        SourceType.SOCIAL_XIAOHONGSHU: {
            "domains": ["xiaohongshu.com", "xhslink.com"],
            "display_name": "小红书",
        },
        SourceType.SOCIAL_ZHIHU: {
            "domains": ["zhihu.com", "zhuanlan.zhihu.com"],
            "display_name": "知乎",
        },
        SourceType.COMPLAINT_HEIMA: {
            "domains": ["ts.21cn.pro", "ts.21cn.com"],
            "display_name": "黑猫投诉",
        },
        SourceType.WEIXIN_GZH: {
            "domains": ["mp.weixin.qq.com", "weixin.sogou.com"],
            "display_name": "微信公众号",
        },
    }

    def __init__(self):
        self._searcher = Searcher()

    def name(self) -> str: return "Site搜索"
    def source_type(self) -> SourceType: return SourceType.GENERAL_WEB  # 动态设置

    def is_available(self) -> bool: return True  # 始终可用

    async def search(
        self,
        query: str,
        source_types: Optional[list[SourceType]] = None,
        **kwargs
    ) -> list[DataPoint]:
        """
        按指定平台类型执行 site: 搜索

        Args:
            query: 搜索关键词
            source_types: 要搜索的平台类型（None = 全部）
        """
        types_to_search = source_types or list(self.SITE_CONFIGS.keys())
        results = []

        for st in types_to_search:
            if st not in self.SITE_CONFIGS:
                continue
            config = self.SITE_CONFIGS[st]
            for domain in config["domains"]:
                try:
                    search_results = self._searcher.site_search(query, domain, max_results=3)
                    for sr in search_results:
                        results.append(DataPoint(
                            title=sr.title,
                            content=sr.snippet,
                            url=sr.url,
                            source_type=st,
                            source_name=config["display_name"],
                            confidence="medium" if sr.snippet else "low",
                        ))
                except Exception as e:
                    log.warning(f"site:搜索失败 [{config['display_name']}]: {e}")

        log.info(f"site:搜索完成: {len(results)} 条结果 (types={[t.value for t in types_to_search]})")
        return results
```

```python
# app/recruitment_source.py
"""
招聘数据聚合器
数据来源：
  1. DuckDuckGo site: 限定搜索（zhipin.com / lagou.com / liepin.com）
  2. 公司官网招聘页（careers 页面定向抓取）

用途：反推战略方向（04 §四第5条："招出海运营=要出海"）
"""

from dataclasses import dataclass
from typing import Optional, List
from app.searcher import Searcher
from app.chinese_sources import DataPoint, SourceType
from app.logger import log

@dataclass
class RecruitmentData(DataPoint):
    """招聘数据点（扩展 DataPoint）"""
    position: str = ""           # 职位名称
    department: str = ""         # 部门
    company: str = ""            # 公司名称
    date: Optional[str] = None   # 发布日期
    skills_hint: Optional[str] = None  # 技能线索
    source_url: Optional[str] = None

class RecruitmentSource:
    """招聘数据聚合器"""

    RECRUITMENT_DOMAINS = {
        "zhipin.com": "Boss直聘",
        "lagou.com": "拉勾",
        "liepin.com": "猎聘",
        "zhaopin.com": "智联招聘",
    }

    CAREERS_PATHS = ["/careers", "/jobs", "/join-us", "/hiring", "/about/careers"]

    def __init__(self):
        self._searcher = Searcher()

    async def search_company(self, company_name: str, max_results: int = 10) -> List[RecruitmentData]:
        """搜索某公司的招聘信息"""
        results = []

        # 1. site: 招聘网站搜索
        for domain, display_name in self.RECRUITMENT_DOMAINS.items():
            try:
                search_results = self._searcher.site_search(
                    f"{company_name} 招聘", domain, max_results=3
                )
                for sr in search_results:
                    results.append(RecruitmentData(
                        title=sr.title,
                        content=sr.snippet,
                        url=sr.url,
                        source_type=SourceType.RECRUITMENT,
                        source_name=display_name,
                        company=company_name,
                        source_url=sr.url,
                    ))
            except Exception as e:
                log.warning(f"招聘搜索失败 [{display_name}]: {e}")

        # 2. 官网 careers 页面（可选，在 fetcher 层处理）
        # 这里只记录官网 URL，实际抓取由 engine 层协调

        log.info(f"招聘搜索完成: {company_name} → {len(results)} 条")
        return results

    def analyze_strategy_signals(self, data: List[RecruitmentData]) -> dict:
        """从招聘数据提炼战略信号"""
        if not data:
            return {"signal": "no_data", "message": "未找到招聘信息"}

        departments = []
        skills = []
        geo_expansion = False

        for item in data:
            # 简单关键词匹配（MVP 版本，后续可升级为 LLM 分析）
            content_lower = (item.content + " " + item.title).lower()
            if "海外" in content_lower or "出海" in content_lower or "overseas" in content_lower:
                geo_expansion = True
            if "ai" in content_lower or "算法" in content_lower or "algorithm" in content_lower:
                skills.append("AI/算法")
            if "销售" in content_lower or "sales" in content_lower:
                departments.append("销售")
            if "运营" in content_lower or "operation" in content_lower:
                departments.append("运营")

        return {
            "signal": "detected",
            "total_positions": len(data),
            "expanding_depts": list(set(departments)),
            "new_skills": list(set(skills)),
            "geo_expansion": geo_expansion,
            "message": self._format_signal_message(geo_expansion, departments, skills),
        }

    def _format_signal_message(self, geo: bool, depts: list, skills: list) -> str:
        parts = []
        if geo:
            parts.append("⚠️ 检测到海外/出海相关招聘，可能正在拓展国际市场")
        if depts:
            parts.append(f"📊 正在扩张部门：{'、'.join(depts)}")
        if skills:
            parts.append(f"💡 技能重点：{'、'.join(skills)}")
        return "\n".join(parts) if parts else "未检测到明显战略信号"
```

```python
# app/scene_search.py
"""
场景感知搜索策略引擎 —— 4 个场景的差异化搜索编排

核心逻辑：
  1. 根据场景选择搜索策略（关键词模板 + 数据源组合 + 权重 + 时效偏置）
  2. 渲染关键词模板（{竞品} → target, {品类} → 从 query 提取）
  3. 并发调用各数据源（通用搜索 + 中文数据源 + 招聘）
  4. 按权重排序 + 去重
  5. 返回统一的 DataPoint 列表
"""

import asyncio
from dataclasses import dataclass, field
from typing import Optional
from app.chinese_sources import DataPoint, SourceType, ChineseSourceManager
from app.searcher import Searcher, SearchResult
from app.recruitment_source import RecruitmentSource
from app.logger import log


@dataclass
class SceneSearchConfig:
    """单个场景的搜索策略配置"""
    scene: str
    keywords_templates: list[str]           # 关键词模板

class SceneSearchEngine:
    """场景感知搜索引擎 —— 对外暴露的统一入口"""

    def __init__(self):
        self.searcher = Searcher()
        self.cn_sources = ChineseSourceManager()
        self.recruitment = RecruitmentSource()

    async def search(
        self,
        scene: str,
        query: str,
        target: Optional[str] = None,
    ) -> list[DataPoint]:
        """
        场景感知搜索主方法

        Args:
            scene: 场景名称（battle_card/pricing/weekly/discovery）
            query: 用户原始 query
            target: 竞品名称（如 "Notion"）

        Returns:
            DataPoint 列表（去重、混合多源）
        """
        if scene not in SCENE_STRATEGIES:
            log.warning(f"未知场景: {scene}, 使用 discovery 作为 fallback")
            scene = "discovery"

        config = SCENE_STRATEGIES[scene]

        # 1. 渲染关键词模板
        category = self._extract_category(query, target)
        queries = self._render_templates(config.keywords_templates, target, category)
        # 加入原始 query
        queries.insert(0, query)

        # 2. 并发调用各数据源
        tasks = []

        # 2a. 通用搜索（用第一个渲染后的 query）
        tasks.append(self._search_general(queries[0]))

        # 2b. 中文数据源（用原始 query + target）
        tasks.append(self._cn_sources.search(
            query if not target else f"{target} {query}",
            source_types=config.source_types,
        ))

        # 2c. 招聘数据（weekly 场景启用）
        if target and SourceType.RECRUITMENT in config.source_types:
            tasks.append(self.recruitment.search_company(target))

        # 3. 汇总结果
        results_lists = await asyncio.gather(*tasks, return_exceptions=True)
        all_data_points: list[DataPoint] = []

        for result in results_lists:
            if isinstance(result, Exception):
                log.error(f"搜索任务异常: {result}")
                continue
            if isinstance(result, list):
                all_data_points.extend(result)

        # 4. 去重（按 URL）
        seen_urls = set()
        unique = []
        for dp in all_data_points:
            key = dp.url or dp.title
            if key not in seen_urls:
                seen_urls.add(key)
                unique.append(dp)

        log.info(
            f"场景[{scene}] 搜索完成: "
            f"query={query[:30]} target={target} → {len(unique)} 条数据点"
        )

        return unique

    # ===== 内部方法 =====

    async def _search_general(self, query: str) -> list[DataPoint]:
        """通用搜索 → 转换为 DataPoint"""
        try:
            results = self.searcher.search(query, max_results=settings.search_max_results)
            return [
                DataPoint(
                    title=r.title,
                    content=r.snippet,
                    url=r.url,
                    source_type=SourceType.GENERAL_WEB,
                    source_name="网页搜索",
                    confidence="high" if r.relevance_score > 0.7 else "medium",
                )
                for r in results
            ]
        except Exception as e:
            log.error(f"通用搜索失败: {query} - {e}")
            return []

    def _render_templates(
        self,
        templates: list[str],
        target: Optional[str],
        category: Optional[str],
    ) -> list[str]:
        """渲染关键词模板"""
        queries = []
        for t in templates:
            q = t.replace("{target}", target or "").replace("{category}", category or "")
            q = q.strip()
            if q:
                queries.append(q)
        return queries

    def _extract_category(self, query: str, target: Optional[str]) -> Optional[str]:
        """从 query 中提取品类（简单启发式）"""
        if target and target in query:
            return query.replace(target, "").strip().rstrip("vs对比分析竞品").strip() or None
        return None
```

```python
# app/models.py
"""
统一数据模型 —— 多源数据的结构化表示

核心模型：
  - Source：单条来源引用（展示给用户看）
  - AnalysisResult：一次分析的完整结果（引擎输出）
  - CompareResult：多竞品对比结果（W3 用）
  - CoverageSummary：数据源覆盖率统计
"""

from pydantic import BaseModel, Field
from typing import Optional
from datetime import datetime
from .chinese_sources import DataPoint, SourceType


class Source(BaseModel):
    """单条来源引用（用于展示和分析追溯）"""
    url: str = ""
    title: str
    snippet: str = ""                       # 引用的原文片段（前 200 字）
    source_type: SourceType = SourceType.GENERAL_WEB
    source_display: str = ""                 # 显示名称
    relevance: str = "high"                  # high / medium / low


class CoverageSummary(BaseModel):
    """数据源覆盖率统计"""
    total_data_points: int = 0
    by_type: dict = Field(default_factory=dict)   # {source_type.value: count}
    has_chinese_source: bool = False              # 是否有中文数据源
    has_commercial_data: bool = False             # 是否有商业结构化数据
    has_recruitment_data: bool = False            # 是否有招聘数据
    coverage_percent: float = 0.0                 # 覆盖率百分比（0-100）

    def calculate(self, data_points: list[DataPoint], total_types: int = 21):
        """计算覆盖率"""
        self.total_data_points = len(data_points)
        covered_types = set()
        for dp in data_points:
            t = dp.source_type.value
            self.by_type[t] = self.by_type.get(t, 0) + 1
            covered_types.add(t)
            if dp.source_type in (
                SourceType.SOCIAL_XIAOHONGSHU, SourceType.SOCIAL_ZHIHU,
                SourceType.SOCIAL_DOUYIN, SourceType.WEIXIN_GZH,
                SourceType.COMPLAINT_HEIMA,
            ):
                self.has_chinese_source = True
            if dp.source_type in (SourceType.COMMERCIAL_TIANYANCHA, SourceType.COMMERCIAL_QIMAI):
                self.has_commercial_data = True
            if dp.source_type == SourceType.RECRUITMENT:
                self.has_recruitment_data = True

        self.coverage_percent = round(len(covered_types) / total_types * 100, 1)


class AnalysisResult(BaseModel):
    """一次分析的完整结果"""
    scene: str
    query: str
    target: Optional[str] = None
    analysis: str = ""                         # LLM 生成的分析文本
    sources: list[Source] = []                 # 来源列表（展示用）
    data_points: list[DataPoint] = []          # 原始数据点（调试/审计用）
    confidence: str = "medium"                 # high / medium / low
    ts: datetime = Field(default_factory=datetime.now)
    coverage: CoverageSummary = Field(default_factory=CoverageSummary)

    def finalize(self):
        """填充覆盖率统计"""
        self.coverage.calculate(self.data_points)


class CompareResult(BaseModel):
    """多竞品对比结果（W3 使用）"""
    targets: list[str] = []                    # 对比的竞品列表
    matrix: dict = Field(default_factory=dict)  # 维度 → {竞品: 值}
    insights: list[str] = []                   # 差异化洞察
    sources: list[Source] = []
```

#### 3.8.1 改造后的 engine.py

```python
# app/engine.py
"""
分析引擎 v2 —— 基于真实多源数据的竞品分析

改造前：用户 query → 加载 prompt → 直接丢 LLM → 返回纯文本
改造后：用户 query → 场景感知搜索 → 内容抓取 → 组装上下文 → LLM → 结构化结果

关键变化：
  - 输入：纯文本 → 多源 DataPoint 列表
  - 输出：str → AnalysisResult（含 sources / data_points / coverage / confidence）
  - Prompt 注入：追加参考资料区块 + 红线约束指令
  - 降级策略：分级降级（逐源失败/中文缺失/全失败）
"""

import os
import asyncio
from openai import OpenAI
from app.config import settings
from app.models import AnalysisResult, Source, CoverageSummary
from app.chinese_sources import DataPoint, SourceType
from app.scene_search import SceneSearchEngine
from app.fetcher import Fetcher
from app.logger import log

# 场景 → 提示词文件映射
SCENES = {
    "battle_card": "battle_card.txt",
    "pricing": "pricing.txt",
    "weekly": "weekly.txt",
    "discovery": "discovery.txt",
}

PROMPTS_DIR = os.path.join(os.path.dirname(__file__), "..", "validation", "prompts")


def _read(prompt_file: str) -> str:
    """读取提示词文件（唯一真相来源）"""
    path = os.path.join(PROMPTS_DIR, prompt_file)
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def _build_reference_context(data_points: list[DataPoint], max_chars: int = 4000) -> str:
    """
    将多源 DataPoint 组装成结构化参考资料区块

    格式：
      ### 1. [标题](url) — 来源：XXX
      > 内容摘要...
    """
    if not data_points:
        return "（未能获取实时参考数据）"

    sections = []
    # 按来源类型分组
    grouped: dict[str, list[DataPoint]] = {}
    for dp in data_points:
        key = dp.source_name or dp.source_type.value
        if key not in grouped:
            grouped[key] = []
        grouped[key].append(dp)

    total_chars = 0
    for source_name, items in grouped.items():
        section_parts = [f"\n### 来自 {source_name}（{len(items)} 条）\n"]
        for i, dp in enumerate(items, 1):
            url_part = f"([{dp.title}]({dp.url})) " if dp.url else ""
            date_part = f" [{dp.published_at}]" if dp.published_at else ""
            content = dp.content[:300] if len(dp.content) > 300 else dp.content
            entry = f"{i}. {url_part}{date_part}\n> {content}\n"
            if total_chars + len(entry) > max_chars:
                section_parts.append(f"\n*（省略 {len(items) - i + 1} 条，已达字数上限）*\n")
                break
            section_parts.append(entry)
            total_chars += len(entry)
        sections.append("".join(section_parts))

    header = f"\n## 参考资料来源（共 {len(data_points)} 条，来自 {len(grouped)} 个数据源）\n"
    return header + "".join(sections)


def _build_system_addons() -> str:
    """追加的系统级指令（红线约束）"""
    return """

## 分析约束（必须遵守）
1. 你的分析必须基于上方提供的「参考资料来源」中的真实数据
2. 每条关键论断必须标注信息来源，格式：[来源：xxx]
3. 对于无法从参考资料中验证的信息，必须明确标注「待核实」
4. 优先使用中文互联网来源的数据（小红书/知乎/公众号/天眼查等）
5. 涉及融资/工商数据时，必须引用天眼查等结构化数据源的数字，不得编造
6. 如果参考资料不足，明确告知用户哪些方面缺乏数据支撑
"""


def _assess_confidence(data_points: list[DataPoint], fetched_count: int) -> str:
    """评估分析置信度"""
    if not data_points:
        return "low"
    has_chinese = any(
        dp.source_type in (
            SourceType.SOCIAL_XIAOHONGSHU, SourceType.SOCIAL_ZHIHU,
            SourceType.COMMERCIAL_TIANYANCHA,
        )
        for dp in data_points
    )
    has_commercial = any(
        dp.source_type in (SourceType.COMMERCIAL_TIANYANCHA, SourceType.COMMERCIAL_QIMAI)
        for dp in data_points
    )
    if has_chinese and has_commercial and fetched_count >= 2:
        return "high"
    if has_chinese or has_commercial or fetched_count >= 2:
        return "medium"
    return "low"


async def analyze(scene: str, query: str, target: str | None = None) -> AnalysisResult:
    """
    主分析方法 —— 多源搜索 → 抓取 → LLM 分析 → 结构化结果

    Args:
        scene: 场景名称（battle_card/pricing/weekly/discovery）
        query: 用户原始 query
        target: 竞品名称（可选）

    Returns:
        AnalysisResult（含 analysis / sources / coverage / confidence）
    """
    result = AnalysisResult(scene=scene, query=query, target=target)

    try:
        # ===== Step 1: 场景感知多源搜索 =====
        search_engine = SceneSearchEngine()
        data_points = await search_engine.search(scene, query, target=target)
        result.data_points = data_points

        # ===== Step 2: 内容抓取（Top N URL） =====
        urls_with_type = [
            (dp.url, dp.source_type)
            for dp in data_points if dp.url
        ][:settings.fetch_max_urls]

        if urls_with_type:
            fetcher = Fetcher()
            pages = await fetcher.fetch([u[0] for u in urls_with_type])
            # 将抓取到的正文补充回 data_points
            url_to_page = {p.url: p for p in pages}
            for dp in data_points:
                if dp.url in url_to_page:
                    fetched = url_to_page[dp.url]
                    if fetched.word_count > len(dp.content):
                        dp.content = fetched.text
                        dp.metadata["fetched_full_text"] = True

        # ===== Step 3: 组装上下文 =====
        context = _build_reference_context(data_points)

        # ===== Step 4: 调 LLM =====
        system_prompt = _read("system.txt")
        scene_prompt = _read(SCENES[scene])
        enhanced_system = system_prompt + "\n\n" + scene_prompt + _build_system_addons()
        enhanced_user = f"{query}\n\n{context}"

        client = OpenAI(
            api_key=settings.llm_api_key,
            base_url=settings.llm_base_url,
        )
        resp = client.chat.completions.create(
            model=settings.model,
            temperature=0.4,
            messages=[
                {"role": "system", "content": enhanced_system},
                {"role": "user", "content": enhanced_user},
            ],
            timeout=30,
        )

        # ===== Step 5: 封装结果 =====
        result.analysis = resp.choices[0].message.content or ""
        result.sources = [
            Source(
                url=dp.url or "",
                title=dp.title,
                snippet=dp.content[:200],
                source_type=dp.source_type,
                source_display=dp.source_name,
                relevance="high" if dp.confidence == "high" else "medium",
            )
            for dp in data_points if dp.content
        ]
        result.confidence = _assess_confidence(data_points, len(pages) if urls_with_type else 0)
        result.finalize()

        log.info(
            f"分析完成: scene={scene} target={target} "
            f"sources={len(result.sources)} confidence={result.confidence} "
            f"coverage={result.coverage.coverage_percent}%"
        )

    except Exception as e:
        log.error(f"分析异常: scene={scene} query={query[:30]} - {e}", exc_info=True)
        # 优雅降级：回退到纯 LLM
        result.analysis = _fallback_analysis(scene, query, target, str(e))
        result.confidence = "low"
        result.finalize()

    return result


def _fallback_analysis(scene: str, query: str, target: str | None, error: str) -> str:
    """降级分析：纯 LLM 模式（无实时数据）"""
    try:
        system_prompt = _read("system.txt") + _read(SCENES[scene])
        system_prompt += "\n\n注意：当前无法获取实时数据，请基于你的知识回答，并明确标注「以下内容待核实」。"

        client = OpenAI(
            api_key=settings.llm_api_key,
            base_url=settings.llm_base_url,
        )
        resp = client.chat.completions.create(
            model=settings.model,
            temperature=0.4,
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": query},
            ],
            timeout=30,
        )
        return resp.choices[0].message.content or f"⚠️ 分析服务暂时不可用（{error}）。请稍后重试。"
    except Exception as e2:
        return f"⚠️ 分析服务暂时不可用。错误：{error}\n我们将尽快恢复服务。"
```

```python
async def _async_analyze(self, scene: str, query: str, target=None):
    # 旧代码：
    # result = analyze(scene, query)  # 返回 str
    # return result

    # 新代码：
    from app.engine import analyze
    result = await analyze(scene, query, target=target)  # 返回 AnalysisResult

    # 格式化展示
    reply = self._format_analysis_result(result)
    return reply

def _format_analysis_result(self, result) -> str:
    """将 AnalysisResult 格式化为飞书文本消息"""
    lines = [result.analysis]

    # 来源分组显示
    if result.sources:
        # 按 source_display 分组
        grouped: dict[str, list] = {}
        for s in result.sources:
            key = s.source_display or s.source_type.value
            if key not in grouped:
                grouped[key] = []
            grouped[key].append(s)

        lines.append("")
        lines.append(f"📚 数据来源（{len(result.sources)} 条）")
        emoji_map = {
            "网页搜索": "🔵", "小红书": "🔴", "知乎": "🟤",
            "天眼查": "🟢", "黑猫投诉": "🟡", "微信公众号": "🟣",
        }
        for name, sources in grouped.items():
            emoji = emoji_map.get(name, "⚪")
            lines.append(f"  {emoji} {name}（{len(sources)} 条）")
            for s in sources[:3]:  # 每类最多显示 3 条
                lines.append(f"    · [{s.title}]({s.url})" if s.url else f"    · {s.title}")

    # 置信度和覆盖率
    conf_emoji = {"high": "✅", "medium": "⚠️", "low": "❓"}
    lines.append("")
    lines.append(f"{conf_emoji.get(result.confidence, '')} 置信度：{result.confidence}")
    if result.coverage.coverage_percent > 0:
        lines.append(f"📊 数据源覆盖率：{result.coverage.coverage_percent}%")

    return "\n".join(lines)
```

## 5. 常见问题与排错

| 问题 | 可能原因 | 解决方法 |
|---|---|---|
| `duckduckgo-search` 安装失败 | Python 版本兼容性 | 用 `pip install duckduckgo-search==7.0.0` 指定版本 |
| 天眼查 API 返回 401 | API Key 无效或过期 | 去 open-api.tianyancha.com 重新生成 |
| site: 搜索结果为空 | DDG 对中文站点索引有限 | 正常现象，后续接第三方 API 补齐 |
| trafilatura 提取空白 | 页面是 JS 渲染（SPA） | 用 Jina Reader fallback（`https://r.jina.ai/{url}`） |
| LLM 超时 | DeepSeek API 延迟或网络问题 | 已有 30s 超时兜底，检查网络 |
| analyze() 整体耗时 >20s | 搜索/抓取串行执行 | 检查是否用了 asyncio.gather 并行 |
